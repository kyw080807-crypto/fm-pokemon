# FM gimmick items (FM::Gimmick): Max Mushrooms and Tera Crystals let a mon Dynamax or Terastallize
# in games that use those gimmicks (the Poké Manager rules). Held, no battle effect of their own.
#-------------------------------
[MAXMUSHROOM]
Name = Max Mushroom
NamePlural = Max Mushrooms
Pocket = 1
Price = 25000
Description = A held item. Lets the holder Dynamax in games where Dynamax is used.

Name = G-Max Mushroom
